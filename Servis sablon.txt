 public class PatientService : IPatientService
 {
     private readonly IPatientRepository _repository;
     private readonly IMapper _mapper;
     public PatientService(IPatientRepository repository, IMapper mapper)
     {
         _repository = repository;
         _mapper = mapper;
     }

     public async Task<Patient?> GetById(int patientId)
     {
         return await _repository.GetById(patientId);
     }

     public async Task<List<Patient>> GetAll()
     {
         return await _repository.GetAll();
     }

     public async Task<Patient?> Update(UpdatePatientDto patientDto, int patientId)
     {
         var patient = await _repository.GetById(patientId);

         if (patient == null)
             return null;

         _mapper.Map(patientDto, patient); // map ON TOP of existing entity

         return await _repository.UpdateAsync(patient, patientId);
     }

     public async Task Delete(int patientId)
     {
         await _repository.DeleteAsync(patientId);
     }

     public async Task Create(CreatePatientDto dto)
     {
         var patient = _mapper.Map<Patient>(dto);

         await _repository.CreateAsync(patient);
     }
 }
